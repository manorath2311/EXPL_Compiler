decl
    int m, n, p;
    int i, j, k;
    int A[10][10], B[10][10], C[10][10];
enddecl;
begin

    read(m);

    read(n);
 
    read(p);

 
    i = 0;
    while (i < m) do
        j = 0;
        while (j < n) do
            read(A[i][j]);
            j = j + 1;
        endwhile;
        i = i + 1;
    endwhile;

 
    i = 0;
    while (i < n) do
        j = 0;
        while (j < p) do
            read(B[i][j]);
            j = j + 1;
        endwhile;
        i = i + 1;
    endwhile;

    i = 0;
    while (i < m) do
        j = 0;
        while (j < p) do
            C[i][j] = 0;
            k = 0;
            while (k < n) do
                C[i][j] = C[i][j] + (A[i][k] * B[k][j]);
                k = k + 1;
            endwhile;
            j = j + 1;
        endwhile;
        i = i + 1;
    endwhile;

   
    i = 0;
    while (i < m) do
        j = 0;
        while (j < p) do
            write(C[i][j]);
            j = j + 1;
        endwhile;
        i = i + 1;
    endwhile;
end;
